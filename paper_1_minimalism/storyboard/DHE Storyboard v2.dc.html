Одно ядро: хеши, нормировка и первый слой сразу

aᵢ, bᵢ и промежуточные значения живут в регистрах · в HBM уходит только выход

@triton.jit
def dhe_l1(E, A, B, W1, Y, K: tl.constexpr,
N: tl.constexpr, BK: tl.constexpr,
BN: tl.constexpr):
row = tl.program_id(0); e = tl.load(E + row)
acc = tl.zeros((BN,), dtype=tl.float32)
for k0 in range(0, K, BK):
kk = k0 + tl.arange(0, BK)
a = tl.load(A + kk); b = tl.load(B + kk)
h = (a * e + b) % P % M # регистры
v = 2.0 * h.to(tl.float32) / (M-1) - 1.0
w = tl.load(W1 + kk[:,None]*N + tl.arange(0,BN))
acc += tl.sum(v[:,None] * w, axis=0)
tl.store(Y + row*N + tl.arange(0,BN), acc)
